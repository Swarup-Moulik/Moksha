void test_dynamic_mutations() {
    println("--- Testing Dynamic Heap Mutations ---");
    int[] arr = [];

    // Push & Pop
    push(arr, 10);
    push(arr, 20);
    push(arr, 30);

    if (length(arr) == 3) {
        println("PASS: push() updates length");
    } else {
        println("[CRITICAL] FAIL: push() updates length");
    }

    int last = pop(arr);

    if (last == 30) {
        println("PASS: pop() returns correct value");
    } else {
        println("[CRITICAL] FAIL: pop() returns correct value");
    }

    if (length(arr) == 2) {
        println("PASS: pop() decrements length");
    } else {
        println("[CRITICAL] FAIL: pop() decrements length");
    }

    // Insert & Remove
    insert(arr, 1, 15); // [10, 15, 20]

    if (arr[1] == 15) {
        println("PASS: insert() shifts elements correctly");
    } else {
        println("[CRITICAL] FAIL: insert() shifts elements correctly");
    }

    int removed = remove(arr, 0); // [15, 20]

    if (removed == 10) {
        println("PASS: remove() returns correct value");
    } else {
        println("[CRITICAL] FAIL: remove() returns correct value");
    }

    if (length(arr) == 2) {
        println("PASS: remove() shifts elements left");
    } else {
        println("[CRITICAL] FAIL: remove() shifts elements left");
    }

    // Extend
    int[] more = [...arr, 100, 200];

    if (length(more) == 4) {
        println("PASS: spread operator merges slices");
    } else {
        println("[CRITICAL] FAIL: spread operator merges slices");
    }

    // Clear & Resize
    clear(arr);

    if (is_empty(arr) == true) {
        println("PASS: clear() resets length");
    } else {
        println("[CRITICAL] FAIL: clear() resets length");
    }

    resize(arr, 5);

    if (length(arr) == 5) {
        println("PASS: resize() allocates exact length");
    } else {
        println("[CRITICAL] FAIL: resize() allocates exact length");
    }
}

void test_search_and_sort() {

        println("PASS: slice() offsets base pointer correctly");
    } else {
        println("[CRITICAL] FAIL: slice() offsets base pointer correctly");
    }
}

void test_calloc_zero_init() {
    println("Running test_calloc_zero_init...")
    unsafe {
        // Allocate an array of 5 integers, zero-initialized
        int count = 5
        *mut int arr = cast<*mut int>(calloc(count, sizeof(int)))

        // Verify zero initialization on the first element
        if (*arr == 0) {
            println("  [PASS] calloc successfully zero-initialized memory")
        } else {
            println("  [FAIL] calloc memory was not zeroed")
        }

        // Pointer arithmetic to check and mutate the second element
        *mut int second_elem = arr + 1
        *second_elem = 100

        if (*(arr + 1) == 100) {
            println("  [PASS] pointer arithmetic successful")
        } else {
            println("  [FAIL] pointer arithmetic failed")
        }

        free(cast<*void>(arr))
    }
}

void test_realloc_growth() {
    println("Running test_realloc_growth...")
    unsafe {
        // Initial allocation for 2 integers
        *mut int ptr = cast<*mut int>(malloc(sizeof(int) * 2))
        *ptr = 10
        *(ptr + 1) = 20

        // Reallocate to hold 4 integers
        int new_size = sizeof(int) * 4
        *mut int grown_ptr = cast<*mut int>(realloc(cast<*void>(ptr), new_size))

        // Verify old data was preserved during the reallocation move
        if (*grown_ptr == 10 && *(grown_ptr + 1) == 20) {
            println("  [PASS] realloc preserved original data")
        } else {
            println("  [FAIL] realloc corrupted original data")
        }

        // Write to the newly expanded space
        *(grown_ptr + 2) = 30
        *(grown_ptr + 3) = 40

        free(cast<*void>(grown_ptr))
    }
}

void test_pointer_to_pointer() {
    println("Running test_pointer_to_pointer...")
    unsafe {
        // Allocate a pointer that points to another pointer (double indirection)
        *mut *mut int handle = cast<*mut *mut int>(malloc(sizeof(usize)))

        // Allocate the actual integer payload
        *handle = cast<*mut int>(malloc(sizeof(int)))

        // Dereference twice to mutate the value
        *(*handle) = 99

        if (*(*handle) == 99) {
            println("  [PASS] Double indirection (**ptr) works")
        } else {
            println("  [FAIL] Double indirection failed")
        }

        free(cast<*void>(*handle)) // Free inner payload
        free(cast<*void>(handle))  // Free outer handle
    }
}

void test_linked_list_nodes() {
    println("Running test_linked_list_nodes...")
    unsafe {
        // Allocate two nodes independently
        *mut Node head = cast<*mut Node>(malloc(sizeof(Node)))
        *mut Node second = cast<*mut Node>(malloc(sizeof(Node)))

        // Link them up
        (*head).data = 100
        (*head).next = second

        // Terminate the list with a null pointer
        (*second).data = 200
        (*second).next = cast<*mut Node>(null)

        // Traverse the pointer chain to check data
        if ((*head).data == 100 && (*((*head).next)).data == 200) {
            println("  [PASS] Linked list struct chaining and traversal")
        } else {
            println("  [FAIL] Linked list traversal failed")
        }

        // Clean up in reverse
        free(cast<*void>(second))
        free(cast<*void>(head))
    }
}
